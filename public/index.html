const express = require('express'); const path = require('path'); const fs = require('fs'); const { default: makeWASocket, useMultiFileAuthState, fetchLatestBaileysVersion } = require('@whiskeysockets/baileys'); const { Boom } = require('@hapi/boom'); const app = express(); const PORT = process.env.PORT || 3000; app.use(express.static('public')); let sock, code; // Make sure session folder exists if (!fs.existsSync('./session')) fs.mkdirSync('./session'); async function connectToWhatsApp() { const { state, saveCreds } = await useMultiFileAuthState('./session'); const { version } = await fetchLatestBaileysVersion(); sock = makeWASocket({ version, auth: state, printQRInTerminal: false, logger: require('pino')({ level: 'silent' }) }); sock.ev.on('creds.update', saveCreds); } connectToWhatsApp(); app.get('/code', async (req, res) => { const number = req.query.number; if(!number) return res.json({error: 'Enter number with country code'}); try { // Wait 2 seconds for socket to be ready await new Promise(r => setTimeout(r, 2000)); code = await sock.requestPairingCode(number); console.log(`🔥 PAIRING CODE: ${code}`); res.json({code: code.match(/.{1,3}/g).join('-')}); } catch (err) { console.error(err); res.json({error: 'Failed. Try again in 10s'}); } }); app.listen(PORT, '0.0.0.0', () => console.log(`✅ Running on ${PORT}`));
